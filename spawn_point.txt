--@name Spawn point
--@author Kid benglao
--@shared
if SERVER then
	hook.add("Think","heal",function()
		local vida = owner():getHealth()
		local batteria = owner():getArmor()
		if vida < 200 then
			owner():setHealth(vida + 1)
		end
		if batteria < 200 then
			owner():setArmor(batteria + 1)
		end
	end)
	local pos
	local holo = hologram.create(Vector(0,0,0) + Vector(0,0,30),Angle(0,0,0),"models/props_c17/gravestone_cross001a.mdl",Vector(0.4,0.4,0.4))
	hook.add("PlayerDeath", "pos", function(ply, inflictor, attacker)
		if ply == owner() then
			pos = ply:getPos()
			holo:setPos(pos + Vector(0,0,30))
			net.start("sound")
            net.send()
			attacker:kill()
		end
	end)
	hook.add("PlayerHurt", "pos", function(ply, attacker, newHealth, damageTaken)
		if ply ~= owner() then return end
		if attacker:isPlayer() and attacker ~= owner() then
			attacker:applyDamage(
				damageTaken,
				ply
			)
		end
	end)

	hook.add("PlayerSpawn","A",function(ply)
		if ply == owner() then
			ply:setPos(pos)
			holo:setPos(Vector(0,0,0))
			net.start("sound2")
            net.send()
		end
	end)
else 
	net.receive("sound", function()
		bass.loadURL("https://cdn.discordapp.com/attachments/1525311386463375481/1552334439923716116/tmp_7901-951678082_3.mp3?ex=6ab53b9b&is=6ab3ea1b&hm=cdc83bb81f636da2d52d2d27c3dff4036c3c582be5cdac563a74267d3daf19a3&", "3d", function(som, erroId, erroNome)

			if not isValid(som) then
				print("Erro no audio" .. erroNome)
				return
			end

			som:play()

			som:setVolume(1) 
			som:setPos(owner():getPos())
		end)
	end)
	net.receive("sound2", function()
		bass.loadURL("https://cdn.discordapp.com/attachments/1525311386463375481/1552337307925872651/roblox-spawn.mp3?ex=6ab53e47&is=6ab3ecc7&hm=33716c87336ea5f1398554df7869a46687048f5d665d653afa26239ccb377b78&", "3d", function(som, erroId, erroNome)

			if not isValid(som) then
				print("Erro no audio" .. erroNome)
				return
			end

			som:play()

			som:setVolume(1) 
			som:setPos(owner():getPos())
		end)
	end)
end
