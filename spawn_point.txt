--@shared
if SERVER then
	local pos
	local holo = hologram.create(Vector(0,0,0) + Vector(0,0,30),Angle(0,0,0),"models/props_c17/gravestone_cross001a.mdl",Vector(0.4,0.4,0.4))
	hook.add("PlayerDeath", "pos", function(ply, inflictor, attacker)
		if ply == owner() then
			pos = ply:getPos()
			holo:setPos(pos + Vector(0,0,30))
			net.start("sound")
            net.send()
		end
	end)

	hook.add("PlayerSpawn","A",function(ply)
		if ply == owner() then
			ply:setPos(pos)
			holo:setPos(Vector(0,0,0))
			net.start("sound2")
            net.send()
		end
	end)
else 
	net.receive("sound", function()
		bass.loadURL("https://uc652612a60195ec25fb1b0de202.dl.dropboxusercontent.com/cd/0/inline/DIspjydDkkYm8nr_MhfuTh5WJ3LVlbxE-v4WxyFU-_LlmkUOox9FeSyRKCJPcT1dIDaAoZLmk64_EkeNsjdEPvjz9MLzJWVovd3EmckPzvvqrqWfINu-UTCv-epl_15K9ILUZKp9mUJbYIu84Ws_zbwh/file#", "3d", function(som, erroId, erroNome)

			if not isValid(som) then
				print("Erro no audio" .. erroNome)
				return
			end

			som:play()

			som:setVolume(1) 
			som:setPos(owner():getPos())
		end)
	end)
	net.receive("sound2", function()
		bass.loadURL("https://ucb71ad36d4d06f829e868a5c459.dl.dropboxusercontent.com/cd/0/inline/DIv9gegqRyOFYyo1pPF5Hw_6j-tu2ZPqEjfMmOaEcPaPCkqJ-t4So6awZVJKqXU-etg4G7JkGZHYraWQkgsYUD2k2uo1-SG6UNCokDkBUv2lIlDxmngYpl4c3umNl6nqhl81gBtyzS226cXUJD0wKaog/file#", "3d", function(som, erroId, erroNome)

			if not isValid(som) then
				print("Erro no audio" .. erroNome)
				return
			end

			som:play()

			som:setVolume(1) 
			som:setPos(owner():getPos())
		end)
	end)
end
